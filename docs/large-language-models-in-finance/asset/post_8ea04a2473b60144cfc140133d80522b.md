```python
ndcg: float

class RAGEvaluator:
    """Comprehensive RAG evaluation."""

    def __init__(self, rag_system):
        self.rag = rag_system

    @staticmethod
    def extract_doc_id(source: Dict) -> Optional[str]:
        """Extract a stable document or chunk identifier from a source."""
        metadata = source.get('metadata', {})
        return metadata.get('chunk_id') or metadata.get('doc_id') or source.get('id')

    def evaluate(self, test_cases: List[Dict], k: int = 5) -> EvaluationResult:
        """Evaluate RAG system on labeled test cases."""
        metrics = {
            'precision': [], 'recall': [], 'mrr': [],
            'ndcg': [], 'faithfulness': [],
            'relevancy': [], 'context_relevancy': [],
            'latency': []
        }

        for test_case in test_cases:
            start = time.perf_counter()
            results = self.rag.query(test_case['query'], n_results=k)
            elapsed_ms = (time.perf_counter() - start) * 1000

            sources = results.get('sources', [])
            retrieved_ids = [
                doc_id for doc_id in (self.extract_doc_id(src)
                for src in sources)
                if doc_id is not None
            ]
            relevant_ids = set(test_case.get('relevant_ids', []))

            relevant_retrieved = set(retrieved_ids) & relevant_ids
            metrics['precision'].append(len(relevant_retrieved) / max(k, 1))
            metrics['recall'].append(
                len(relevant_retrieved) / max(len(relevant_ids), 1)
            )

            reciprocal_rank = 0.0
            for rank, doc_id in enumerate(retrieved_ids, 1):
                if doc_id in relevant_ids:
                    reciprocal_rank = 1.0 / rank
                    break
            metrics['mrr'].append(reciprocal_rank)

            metrics['ndcg'].append(self.compute_ndcg(retrieved_ids,
                relevant_ids, k))
            metrics['faithfulness'].append(
                self.compute_faithfulness(results['answer'], sources)
            )
            metrics['relevancy'].append(
                self.compute_relevancy(test_case['query'], results['answer'])
            )
            metrics['context_relevancy'].append(
                self.compute_context_relevancy(test_case['query'], sources)
            )
            metrics['latency'].append(elapsed_ms)

        return EvaluationResult(
            retrieval_precision=float(np.mean(metrics['precision'])),
            retrieval_recall=float(np.mean(metrics['recall'])),
            mrr=float(np.mean(metrics['mrr'])),
            ndcg=float(np.mean(metrics['ndcg'])),
            faithfulness=float(np.mean(metrics['faithfulness'])),
            answer_relevancy=float(np.mean(metrics['relevancy'])),
            context_relevancy=float(
                np.mean(metrics['context_relevancy'])
            ),
            latency_ms=float(np.mean(metrics['latency']))
        )

    def compute_ndcg(
        self, retrieved_ids: List[str], relevant_ids: set, k: int
    ) -> float:
        """Compute binary NDCG@k."""
        dcg = 0.0
        for rank, doc_id in enumerate(retrieved_ids[:k], 1):
            if doc_id in relevant_ids:
                dcg += 1.0 / np.log2(rank + 1)
        ideal_hits = min(len(relevant_ids), k)
        idcg = sum(1.0 / np.log2(rank + 1)
        for rank in range(1, ideal_hits + 1))
        return dcg / idcg if idcg > 0 else 0.0

    @staticmethod
    def _tokens(text: str) -> set:
        """Small deterministic tokenizer for smoke-test metrics."""
        return {
        tok for tok in re.findall(r"[A-Za-z0-9$%]+", text.lower())
        if len(tok) > 2
    }

    def compute_faithfulness(self, answer: str, sources: List[Dict]) -> float:
        """Lexical support proxy for CI smoke tests.

        Production deployments should replace this with a claim-level
        judge, but this deterministic proxy prevents the evaluator from
        silently returning zeros.
        """
        context = " ".join(
            src.get('document', '') or src.get('text', '') for src in sources
        )
        context_tokens = self._tokens(context)
        sentences = [
            s.strip() for s in re.split(r"(?<=[.!?])\s+", answer)
            if s.strip()
        ]
        if not sentences or not context_tokens:
            return 0.0

        support = []
        for sentence in sentences:
            sentence_tokens = self._tokens(sentence)
            support.append(
                len(sentence_tokens & context_tokens) / max(len(sentence_tokens), 1)
            )

        return float(np.mean(support))

    def compute_relevancy(self, query: str, answer: str) -> float:
        """Simple query-answer token-overlap relevancy proxy."""
        q_tokens = self._tokens(query)
        a_tokens = self._tokens(answer)
        return len(q_tokens & a_tokens) / max(len(q_tokens), 1)

    def compute_context_relevancy(
        self, query: str, sources: List[Dict]
    ) -> float:
        """Simple query-context token-overlap relevancy proxy."""
        q_tokens = self._tokens(query)
        context = " ".join(
            src.get('document', '') or src.get('text', '') for src in sources
        )
        c_tokens = self._tokens(context)
        return len(q_tokens & c_tokens) / max(len(q_tokens), 1)
```

The structure of the class mirrors the structure of the section: retrieval metrics are computed against the labeled relevant documents for each query, generation metrics are computed by comparing the generated answer to the query and the retrieved context, and the per-query results are then averaged into a single `EvaluationResult`. The lexical scoring functions in the listing are intentionally conservative smoke tests. They are useful in continuous integration because they are deterministic and cheap, but they should not be mistaken for a complete faithfulness audit. In production, the faithfulness hook should usually be replaced with a claim-level checker that decomposes the answer into atomic claims and verifies each claim against the retrieved passages, while relevancy can be strengthened with embedding similarity or expert-labeled calibration data.

## 4.11 Production considerations

A RAG system that performs well on a benchmark and one that performs well in production at a financial institution are rarely the same artifact. The evaluation protocol of the previous section establishes whether a pipeline is good enough to deploy; this section addresses what it takes to keep it good once it is deployed. Three concerns dominate. First, the system must _scale_: an index that comfortably serves a research prototype may fail at the throughput, latency, and cost profile a production application requires. Second, the system must be _observable_: in regulated settings, every query, retrieval result, and generated answer is potentially subject to audit, and silent quality regressions cannot be tolerated. Third, the system must _improve continuously_: corpora drift, user expectations evolve, and embedding models are routinely upgraded, so a static deployment quickly becomes a stale one. We address each of these in turn.

### 4.11.1 Scaling RAG systems

The first set of pressures comes from sheer volume. As a financial RAG system absorbs more documents, more users, and more concurrent workloads, several scaling challenges surface that the basic pipeline does not address.

- 1. **Index size**: Millions of documents → billions of embeddings
- 2. **Query throughput**: Thousands of queries per second
- 3. **Latency**: Sub-second response times required
- 4. **Cost**: Embedding API costs, vector DB costs
- 5. **Freshness**: Continuous document updates

These pressures rarely act in isolation. Adding more documents drives up index size and embedding cost; serving more queries drives up throughput and latency requirements; and continuous corpus updates strain both. The standard production responses—sharding, caching, and batch processing—each target one or more of these axes simultaneously.

**Sharding** distributes documents across multiple shards to keep per-shard index size manageable and to parallelise queries across machines:

```
shard(d) = hash(id(d)) mod n_shards
```

Caching, via cache embeddings and frequent queries, converts repeat work into O(1) lookups:

- - Embedding cache: O(1) lookup vs. O(d) computation
- - Query result cache: Avoid redundant retrieval
- - TTL-based invalidation for freshness

Batch processing amortizes fixed overheads—API call setup, model loading, network round-trips—across many items per batch:

Cost per query = \frac{\text{Fixed overhead} + n \cdot \text{Variable cost}}{n} (4.24)

As n \to \infty, cost per query \to \text{variable cost}.

### 4.11.2 Monitoring and observability

Scaling addresses the question of whether a system can handle production load; monitoring addresses whether it is actually doing the right thing once it is there. RAG systems fail in subtle, silent ways—a re-indexing job ships a worse embedding model, a new chunking strategy degrades recall, a prompt change increases hallucination—and none of these are visible without explicit instrumentation. The metrics below are the minimum that any production financial RAG system should track continuously.

Key metrics to track:

- - Latency: P50, P95, P99 retrieval and generation times
- - Quality: Average relevance scores, user feedback ratings
- - Coverage: Fraction of queries with relevant results
- - Cost: API costs, compute costs, storage costs
- - Freshness: Document age distribution in results

In a regulated context, these metrics are not just operational dashboards but governance artifacts: they feed model risk reviews, support audit trails, and underpin the go/no-go thresholds defined in the evaluation protocol.

For financial institutions, monitoring should also record the controls that surrounded the answer: the corpus version, embedding model version, retrieval filters, user entitlement checks, prompt template version, generated citations, reviewer outcome when human approval is required, and any blocked or escalated requests. These fields make it possible to reconstruct why a system answered as it did and to distinguish a retrieval failure from a generation failure or a permissioning failure.
