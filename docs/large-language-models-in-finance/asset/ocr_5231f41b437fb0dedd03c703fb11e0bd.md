The following examples show simple access patterns for retrieving market prices and SEC filings. They are intended for experimentation and should be adapted before use in production pipelines.

The first example downloads historical price data for a small list of tickers and returns a dictionary of data frames.

```
```python
import yfinance as yf
import pandas as pd

def get_market_data(tickers, period="1y", interval="1d"):
    """
    Download historical OHLCV data with error handling
    """
    data = {}
    for ticker in tickers:
        try:
            df = yf.download(ticker, period=period, interval=interval,
                             auto_adjust=True, progress=False)
            data[ticker] = df
        except Exception as e:
            print(f"Error fetching {ticker}: {e}")
            data[ticker] = pd.DataFrame()
    return data

# Example usage
if __name__ == "__main__":
    portfolio = ["AAPL", "MSFT", "JPM", "GS"]
    market_data = get_market_data(portfolio, period="5y")
