def train(self):
    dataset = self.load_and_prepare_data()

    args = TrainingArguments(
        output_dir=self.output_dir,
        eval_strategy="epoch",
        save_strategy="epoch",
        learning_rate=2e-5,
        per_device_train_batch_size=16,
        per_device_eval_batch_size=16,
        num_train_epochs=3,
        weight_decay=0.01,
        logging_steps=50,
        # HI-20. Pin EVERY source of randomness, not just the split.
        # seed governs init and shuffling; data_seed governs the data
        # sampler independently. Leaving either unpinned means the
        # "reproducible" run is reproducible only by luck.
        seed=42,
        data_seed=42,
        load_best_model_at_end=True,
        metric_for_best_model="macro_f1",
    )

    trainer = Trainer(
        model=self.model,
        args=args,
        train_dataset=dataset["train"],
        # Select on VALIDATION, never on test.
        eval_dataset=dataset["validation"],
        tokenizer=self.tokenizer,
        compute_metrics=self.compute_metrics,
    )

    trainer.train()
    trainer.save_model(self.output_dir)
    self.tokenizer.save_pretrained(self.output_dir)
    return trainer
